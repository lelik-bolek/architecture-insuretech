# Нормализация переводов строк
* text=auto eol=lf

# Явно бинарные файлы
*.png  binary
*.jpg  binary
*.jpeg binary
*.gif  binary
*.ico  binary
*.pdf  binary
*.zip  binary
*.jar  binary
*.exe  binary
*.dll  binary

# Windows-скрипты должны оставаться с CRLF
*.bat  text eol=crlf
*.cmd  text eol=crlf
*.ps1  text eol=crlf

# Unix-скрипты — LF
*.sh   text eol=lf

# Файлы, которые Git не должен трогать
*.sln  text eol=crlf
*.vcxproj text eol=crlf
